% !TEX program = xelatex
% Example 2.6: Text alignment
% Chapter 2, Section 2.3

\documentclass[12pt, a4paper]{article}
\usepackage{fontspec}
\setmainfont{Latin Modern Roman}

\begin{document}

\section*{Justified (Default)}
This text is fully justified. Both the left and right margins are aligned.
LaTeX achieves this by stretching spaces between words and hyphenating
when necessary.

\section*{Centered}
\begin{center}
This text is centered. \\
It uses the center environment.
\end{center}

\section*{Left-Aligned}
\begin{flushleft}
This text is left-aligned. \\
The right edge is ragged.
\end{flushleft}

\section*{Right-Aligned}
\begin{flushright}
This text is right-aligned. \\
The left edge is ragged.
\end{flushright}

\section*{Declaration Form}
{\centering Centered with declaration.\par}
{\raggedright Left-aligned with declaration.\par}
{\raggedleft Right-aligned with declaration.\par}

\end{document}
