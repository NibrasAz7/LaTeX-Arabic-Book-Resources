% Example 2.1: Basic fontspec setup
% Chapter 2, Section 2.1

\documentclass[12pt, a4paper]{article}
\usepackage{fontspec}

\setmainfont{TeX Gyre Pagella}
\setsansfont{TeX Gyre Heros}[Scale=MatchLowercase]
\setmonofont{Inconsolata}[Scale=MatchLowercase]

\begin{document}
This text uses TeX Gyre Pagella (serif).

\sffamily This text uses TeX Gyre Heros (sans-serif).

\ttfamily This text uses Inconsolata (monospace).
\end{document}
