\documentclass[12pt, a4paper]{article}
\usepackage{fontspec}
\usepackage{enumitem}
\setmainfont{Latin Modern Roman}

\begin{document}

\section*{Custom Labels}
\begin{enumerate}[label=\textbf{Step \arabic*:}, leftmargin=2.5cm]
  \item Prepare the electrode placement.
  \item Calibrate the recording device.
  \item Record baseline activity.
\end{enumerate}

\section*{Compact List (No Separation)}
\begin{itemize}[nosep, label=$\bullet$]
  \item First item
  \item Second item
  \item Third item
\end{itemize}

\section*{Resume Counting}
\begin{enumerate}
  \item First item in list one
  \item Second item in list one
\end{enumerate}

Some text between lists.

\begin{enumerate}[resume]
  \item Third item (continues from above)
  \item Fourth item
\end{enumerate}

\section*{Custom Spacing}
\begin{itemize}[itemsep=10pt, parsep=0pt, topsep=5pt]
  \item Item with custom spacing
  \item Another item
\end{itemize}

\end{document}
