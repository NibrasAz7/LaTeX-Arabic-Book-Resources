% Example 2.5: Colors with xcolor
% Chapter 2, Section 2.3

\documentclass[12pt, a4paper]{article}
\usepackage{fontspec}
\usepackage{xcolor}
\setmainfont{Latin Modern Roman}

% Define custom colors
\definecolor{universityblue}{HTML}{003366}
\definecolor{alertred}{HTML}{CC0000}
\definecolor{codebg}{rgb}{0.95, 0.95, 0.95}

\begin{document}

\section*{Text Color}
\textcolor{red}{This text is red.} \\
\textcolor{blue}{This text is blue.} \\
\textcolor{universityblue}{University blue text.} \\
\textcolor{alertred}{Alert red text.}

\section*{Background Color}
\colorbox{yellow}{Yellow background text.} \\
\colorbox{codebg}{Code-like background.}

\section*{Color Mixing}
\textcolor{red!50!blue}{50\% red + 50\% blue = purple.} \\
\textcolor{red!20}{20\% red + 80\% white = light red.} \\
\textcolor{red!30!black}{30\% red + 70\% black = dark red.}

\end{document}
