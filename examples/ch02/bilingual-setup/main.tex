\documentclass[12pt, a4paper]{article}
\usepackage{fontspec}
\usepackage{polyglossia}

\setdefaultlanguage[locale=mashriq]{arabic}
\setotherlanguage{english}

\newfontfamily\arabicfont[Script=Arabic, Scale=1.1]{Amiri}
\newfontfamily\englishfont[Script=Latin]{TeX Gyre Pagella}

\begin{document}

% Arabic is the default language (RTL)
% Arabic text goes here, written directly.

% Inline English within Arabic:
% Use \LR{English text} for short inline LTR text.

% English block:
\begin{english}
This entire section is in English, written left-to-right.
Multiple paragraphs can be included here.
\end{english}

% Back to Arabic (RTL)

\end{document}
