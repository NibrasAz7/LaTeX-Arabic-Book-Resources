% Example 2.2: Font types - each name shown in its own typeface
% Chapter 2, Section 2.1

\documentclass[12pt, a4paper]{article}
\usepackage{fontspec}

\setmainfont{Latin Modern Roman}
\setsansfont{Latin Modern Sans}
\setmonofont{Latin Modern Mono}

% Define additional font families for demonstration
\newfontfamily\timesfont{TeX Gyre Termes}      % Times-like serif
\newfontfamily\palatinofont{TeX Gyre Pagella}   % Palatino-like serif
\newfontfamily\helveticafont{TeX Gyre Heros}    % Helvetica-like sans
\newfontfamily\arialfont{TeX Gyre Adventor}     % Arial-like sans
\newfontfamily\courierfont{TeX Gyre Cursor}     % Courier-like mono
\newfontfamily\inconsolatafont{InconsolataN}     % Modern mono

\begin{document}

\section*{Serif Fonts (with serifs)}
{\textrm{Latin Modern Roman}} --- the default font. \\
{\timesfont TeX Gyre Termes} --- Times-like. \\
{\palatinofont TeX Gyre Pagella} --- Palatino-like.

\section*{Sans-Serif Fonts (no serifs)}
{\textsf{Latin Modern Sans}} --- the default sans. \\
{\helveticafont TeX Gyre Heros} --- Helvetica-like. \\
{\arialfont TeX Gyre Adventor} --- Arial-like.

\section*{Monospace Fonts (fixed width)}
{\texttt{Latin Modern Mono}} --- the default mono. \\
{\courierfont TeX Gyre Cursor} --- Courier-like. \\
{\inconsolatafont Inconsolata} --- modern mono.

\end{document}
