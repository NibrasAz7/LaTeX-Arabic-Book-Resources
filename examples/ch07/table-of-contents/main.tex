% !TEX program = xelatex
% Example: Table of contents with tocdepth control
% Demonstrates \tableofcontents with sections and subsections,
% and \setcounter{tocdepth}{2} to show subsections in the TOC.
\documentclass[11pt]{article}
\usepackage{hyperref}

% Show entries down to subsection level (2) in the table of contents
\setcounter{tocdepth}{2}

\begin{document}

\tableofcontents

\section{Introduction}
This section appears in the table of contents.

\subsection{Background}
This subsection also appears because \texttt{tocdepth} is set to 2.

\subsection{Motivation}
Another subsection for demonstration.

\section{Methods}

\subsection{Data Collection}
Details about data collection go here.

\subsubsection{Sources}
Subsubsections do not appear in the TOC at depth 2.

\section{Results}
The results section summarizes findings.

\subsection{Analysis}
Analysis of the results.

\section{Conclusion}
The conclusion wraps up the document.

\end{document}
