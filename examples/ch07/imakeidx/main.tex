% !TEX program = xelatex
% Example: Index with the imakeidx package
% Demonstrates \makeindex, \index entries throughout the text,
% and \printindex to display the compiled index.
\documentclass[11pt]{article}
\usepackage{imakeidx}
\usepackage{hyperref}

% Enable indexing and configure the index style
\makeindex

\begin{document}

\section{Introduction}
This document demonstrates how to build an index.
We mention \index{LaTeX}LaTeX\ and \index{indexing}indexing
as key topics.

\section{Concepts}
The \index{indexing!automatic}automatic indexing approach
uses the \texttt{imakeidx} package.
Subentries are created with the \texttt{!} separator, as in
\index{indexing!subentries}subentries.

We also reference \index{hyperref}\texttt{hyperref} for
clickable cross-references.

\section{Summary}
Multiple \index{indexing!automatic}references to the same
index entry are merged automatically in the final index.

% Print the compiled index
\printindex

\end{document}
