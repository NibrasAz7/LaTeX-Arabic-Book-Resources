% !TEX program = xelatex
% Example: Custom macros with \newcommand
% Demonstrates defining commands with no arguments, one argument,
% and two arguments.
\documentclass[11pt]{article}
\usepackage{hyperref}
\usepackage{amssymb}

% --- No arguments ---
\newcommand{\R}{\mathbb{R}}
\newcommand{\eg}{e.g.,\ }

% --- One argument ---
\newcommand{\vecbold}[1]{\mathbf{#1}}

% --- Two arguments ---
\newcommand{\dotprod}[2]{#1 \cdot #2}

\begin{document}

\section{Custom Commands}

\subsection{No arguments}
The set of real numbers is denoted $\R$.
Use \eg\ when giving an example.

\subsection{One argument}
A bold vector can be written as $\vecbold{x}$ or $\vecbold{v}$.

\subsection{Two arguments}
The dot product of $\vecbold{a}$ and $\vecbold{b}$ is
$\dotprod{\vecbold{a}}{\vecbold{b}}$.

\end{document}
