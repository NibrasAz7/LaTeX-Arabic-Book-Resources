% Example: Comparing built-in Beamer themes
% Uses the Madrid theme as the active example. Commented-out lines show
% how to switch to Warsaw, Berlin, or Singapore to compare the look.

\usetheme{Madrid}

\institute{University of Mathematics}

\begin{document}

\begin{frame}
  \titlepage
\end{frame}

\begin{frame}{Madrid Theme}
  This frame uses the \alert{Madrid} theme:
  \begin{itemize}
    \item Header and footer bars with title and author.
    \item Navigation symbols hidden by default here.
    \item Blue color palette throughout.
  \end{itemize}
  \vspace{0.5em}
  Swap the \texttt{\textbackslash usetheme} line to compare:
  \texttt{Warsaw}, \texttt{Berlin}, or \texttt{Singapore}.
\end{frame}

\begin{frame}{Theme Anatomy}
  Every built-in theme combines:
  \begin{itemize}
    \item An \emph{outer} theme (headers/footers).
    \item An \emph{inner} theme (titles, blocks).
    \item A \emph{color} theme.
    \item A \emph{font} theme.
  \end{itemize}
\end{frame}

\end{document}
