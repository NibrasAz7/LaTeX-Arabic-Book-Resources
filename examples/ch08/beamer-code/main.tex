% !TEX program = xelatex
% Example: Code listings in Beamer
% Demonstrates a [fragile] frame with syntax-highlighted code using
% the listings package. Fragile frames are required for verbatim content.
\documentclass{beamer}
\usetheme{default}
\usepackage{listings}
\usepackage{xcolor}

\lstdefinestyle{pystyle}{
  language=Python,
  basicstyle=\ttfamily\footnotesize,
  keywordstyle=\color{blue}\bfseries,
  commentstyle=\color{green!50!black},
  stringstyle=\color{red!70!black},
  showstringspaces=false,
  numbers=left,
  numberstyle=\tiny\color{gray},
  frame=single,
  breaklines=true,
}

\title{Code in Beamer}
\author{Dr.~Dennis Ritchie}
\institute{Bell Labs}
\date{2025}

\begin{document}

\begin{frame}
  \titlepage
\end{frame}

\begin{frame}[fragile]{A Python Example}
  Frames containing verbatim or listings must be marked \texttt{[fragile]}:
\begin{lstlisting}[style=pystyle]
def fibonacci(n):
    """Return the n-th Fibonacci number."""
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

# Print the first ten numbers
for i in range(10):
    print(fibonacci(i))
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Listings Options}
  Key \texttt{listings} options used above:
  \begin{itemize}
    \item \texttt{language=Python} selects the syntax rules.
    \item \texttt{keywordstyle} colors reserved words.
    \item \texttt{numbers=left} adds line numbers.
    \item \texttt{frame=single} draws a border.
  \end{itemize}
\end{frame}

\end{document}
