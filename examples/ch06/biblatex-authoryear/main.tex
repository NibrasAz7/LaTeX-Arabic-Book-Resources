% BibLaTeX author-year example: demonstrates biblatex with style=authoryear,
% using \textcite (produces "Author (year)") and \parencite (produces "(Author, year)").
% The .bib file is embedded via filecontents* so the document is self-contained.
% Compile with: xelatex main.tex -> biber main -> xelatex main -> xelatex main
\documentclass[11pt]{article}
\usepackage[backend=biber,style=authoryear]{biblatex}
\usepackage{filecontents}
\begin{filecontents*}{refs.bib}
@article{shannon1948,
  author  = {Claude E. Shannon},
  title   = {A Mathematical Theory of Communication},
  journal = {Bell System Technical Journal},
  year    = {1948},
  volume  = {27},
  pages   = {379--423}
}
@book{turing1950,
  author    = {Alan M. Turing},
  title     = {Computing Machinery and Intelligence},
  publisher = {Oxford University Press},
  year      = {1950}
}
\end{filecontents*}
\addbibresource{refs.bib}
\begin{document}

\section*{BibLaTeX Author-Year Example}

This document demonstrates biblatex with the \texttt{authoryear} style.
\textcite{shannon1948} laid the foundation of information theory.
In contrast, \parencite{turing1950} explores machine intelligence.

\printbibliography

\end{document}
