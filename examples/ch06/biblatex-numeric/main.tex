% BibLaTeX numeric example: demonstrates biblatex with style=numeric,
% producing numbered citations like [1], [2]. The .bib file is embedded
% via filecontents* so the document is self-contained.
% Compile with: xelatex main.tex -> biber main -> xelatex main -> xelatex main
\documentclass[11pt]{article}
\usepackage[backend=biber,style=numeric]{biblatex}
\usepackage{filecontents}
\begin{filecontents*}{refs.bib}
@article{turing1936,
  author  = {Alan M. Turing},
  title   = {On Computable Numbers, with an Application to the Entscheidungsproblem},
  journal = {Proceedings of the London Mathematical Society},
  year    = {1936},
  volume  = {42},
  pages   = {230--265}
}
@book{hopcroft1979,
  author    = {John E. Hopcroft and Jeffrey D. Ullman},
  title     = {Introduction to Automata Theory, Languages, and Computation},
  publisher = {Addison-Wesley},
  year      = {1979}
}
\end{filecontents*}
\addbibresource{refs.bib}
\begin{document}

\section*{BibLaTeX Numeric Example}

This document demonstrates biblatex with the \texttt{numeric} style.
Turing's foundational work \cite{turing1936} defined computability.
The textbook by Hopcroft and Ullman \cite{hopcroft1979} covers automata theory.

\printbibliography

\end{document}
