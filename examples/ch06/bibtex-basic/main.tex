% Basic BibTeX example: demonstrates the traditional BibTeX workflow with
% \bibliographystyle{plain} and \bibliography. The .bib file is embedded
% via filecontents* so the document is self-contained.
% Compile with: xelatex main.tex -> bibtex main -> xelatex main -> xelatex main
\documentclass[11pt]{article}
\usepackage{filecontents}
\begin{filecontents*}{refs.bib}
@article{einstein1905,
  author  = {Albert Einstein},
  title   = {On the Electrodynamics of Moving Bodies},
  journal = {Annalen der Physik},
  year    = {1905},
  volume  = {322},
  number  = {10},
  pages   = {891--921}
}
@book{knuth1984,
  author    = {Donald E. Knuth},
  title     = {The {\TeX}book},
  publisher = {Addison-Wesley},
  year      = {1984},
  address   = {Reading, MA}
}
@inproceedings{lamport1994,
  author    = {Leslie Lamport},
  title     = {{\LaTeX}: A Document Preparation System},
  booktitle = {Proceedings of the Annual Meeting on Documentation},
  year      = {1994},
  pages     = {121--132}
}
\end{filecontents*}
\begin{document}

\section*{Basic BibTeX Example}

This document demonstrates the traditional BibTeX workflow.
Einstein's seminal paper \cite{einstein1905} introduced special relativity.
Knuth's book \cite{knuth1984} is the definitive reference on \TeX.
Lamport's contribution \cite{lamport1994} describes \LaTeX{} in detail.

\bibliographystyle{plain}
\bibliography{refs}

\end{document}
