\documentclass[11pt, a4paper]{article}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning}

\begin{document}

\section*{مخططٌ انسيابيٌّ بـ TikZ}

\tikzset{
  startstop/.style={
    rectangle, rounded corners=8pt,
    minimum width=2.5cm, minimum height=1cm,
    text centered, draw, fill=blue!20
  },
  process/.style={
    rectangle,
    minimum width=2.5cm, minimum height=1cm,
    text centered, draw, fill=green!15
  },
  decision/.style={
    diamond, aspect=2,
    minimum width=2cm, minimum height=1cm,
    text centered, draw, fill=yellow!30
  },
  arrow/.style={thick, ->, >=Stealth}
}

\begin{tikzpicture}[node distance=1.5cm]
  % العقد
  \node[startstop] (start) {Start};
  \node[process, below=of start] (input) {Read Input};
  \node[decision, below=of input, yshift=-0.3cm] (check) {Valid?};
  \node[process, below=of check, yshift=-0.3cm] (process) {Process Data};
  \node[startstop, below=of process] (end) {End};

  % الأسهم
  \draw[arrow] (start) -- (input);
  \draw[arrow] (input) -- (check);
  \draw[arrow] (check) -- node[right] {Yes} (process);
  \draw[arrow] (process) -- (end);

  % فرعُ الرفض
  \draw[arrow] (check) -- node[above] {No} ++(3,0) |- (input);
\end{tikzpicture}

\end{document}
