  \zeta \quad \eta \quad \theta \quad \lambda \quad \mu \quad \nu \quad

  \Gamma \quad \Delta \quad \Theta \quad \Lambda \quad \Pi \quad
  \Sigma \quad \Phi \quad \Psi \quad \Omega

\[
  \leq \quad \geq \quad \neq \quad \approx \quad \equiv \quad
  \sim \quad \propto
\]

\[
  \to \quad \rightarrow \quad \leftarrow \quad \leftrightarrow \quad
  \Rightarrow \quad \Leftarrow \quad \Leftrightarrow \quad \mapsto
\]

Set theory:

\[
  \in \quad \notin \quad \subset \quad \subseteq \quad \supset \quad
  \cup \quad \cap \quad \emptyset \quad \forall \quad \exists

Operators and misc:

\[
  \sum \quad \prod \quad \int \quad \oint \quad \infty \quad
  \partial \quad \nabla \quad \pm \quad \mp \quad \times \quad \div
\]
