% Example: Matrices with matrix, pmatrix, bmatrix, vmatrix
% Demonstrates the different matrix delimiters available in amsmath
% and a simple matrix equation.
\documentclass[12pt, a4paper]{article}
\usepackage{amsmath, amssymb, amsthm}
\usepackage{fontspec}
\begin{document}

\section*{Matrix Environments}

Plain matrix (no delimiters):

\[
\begin{matrix}
  1 & 2 & 3 \\
  4 & 5 & 6 \\
  7 & 8 & 9
\end{matrix}
\]

Parentheses with \texttt{pmatrix}:

\[
\begin{pmatrix}
  a_{11} & a_{12} \\
  a_{21} & a_{22}
\end{pmatrix}
\]

Brackets with \texttt{bmatrix}:

\[
\begin{bmatrix}
  1 & 0 \\
  0 & 1
\end{bmatrix}
\]

Determinant with \texttt{vmatrix}:

\[
\begin{vmatrix}
  a & b \\
  c & d
\end{vmatrix}
= ad - bc
\]

A matrix equation:

\[
\begin{pmatrix}
  1 & 2 \\
  3 & 4
\end{pmatrix}
\begin{pmatrix}
  x \\
  y
\end{pmatrix}
=
\begin{pmatrix}
  5 \\
  11
\end{pmatrix}
\]

\end{document}
